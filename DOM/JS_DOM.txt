1. HTML DOM API
----------------------------
<html>
<body>

<p id="demo" title="this is dummy title"></p>

<script>
// Access a paragraph Element
const myPara = document.getElementById("demo");

// Change the content of the Element
myPara.innerHTML = "Hello World!";
myPara.style.color="blue";
myPara.style.fontSize="30px";

myPara.setAttribute("title","setting dummy title attribute");
console.log(myPara.title);

//Create Element
const newPara = document.createElement("p");
newPara.textContent = "This is new para";
newPara.style.color = "red";

//Append Child
document.body.appendChild(newPara);

//Replace Child
const replacePara = document.createElement("p");
replacePara.textContent = "I am replaced para";
replacePara.style.color = "green";

document.body.replaceChild(replacePara, newPara);

//Remove Child
const removePara = document.createElement("p");
removePara.textContent = "I will be removed";

//first append and then remove...
document.body.appendChild(removePara);

document.body.removeChild(removePara);

</script>

</body>
</html>

3. Selecting DOM Elements
-------------------------------------
<html>
<body>

<script>
  //elementById
const elementById = document.getElementById("intro");
document.getElementById("demo").innerHTML = "Jhon Cena!"+elementById.innerHTML;

//elementByClassName
const elementByClassName = document.getElementsByClassName("demoClass");
elementByClassName[0].innerHTML = "Get Element By Class Name";

//elementByClassName
const elementsByTagname = document.getElementsByTagName("p");

7. HTML DOM Animations
-------------------------------
<style>
  #container{
  width:400px;
  height:400px;
  position: relative;
  background-color: yellow;
}
#animate{
  width: 50px;
  height: 50px;
  position: relative;
  background-color: red;
}
</style>

<body>
<p><button onclick="myMove()">Click Me!</button></p>
<div id="container">
  <div id="animate">
  </div>
</div>
</body>

<script>
  function myMove(){
  let id = null;
  let ele = document.getElementById("animate");
  let pos = 0;
  id = setInterval(frame,5);

  function frame()
  {
    if(pos == 350)
    {
      clearInterval(id);
    }
    else{
      pos++;
      ele.style.top = pos+"px";
      ele.style.left = pos+"px";
    }
  }
}
</script>
